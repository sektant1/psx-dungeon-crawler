template< typename TYPE, int SIZE >
class MovingAverage
{
TYPE m_samples[SIZE];
TYPE m_sum;
U32 m_curSample;
U32 m_sampleCount;
public:
MovingAverage() :
m_sum(static_cast<TYPE>(0)),
m_curSample(0),
m_sampleCount(0)
{
}
void addSample(TYPE data)
{
if (m_sampleCount == SIZE)
{
m_sum -= m_samples[m_curSample];
}
else
{
m_sampleCount++;
}
m_samples[m_curSample] = data;
m_sum += data;
m_curSample++;
if (m_curSample >= SIZE)
{
m_curSample = 0;
}
}
